* Functions & Modularity:
  * Documentation & prototypes: header file `.h`
  * Definitions: source file `.c`
* Compiling:
  * Compile without linking: `gcc -c run_utils.c` using the `-c` (for compile) flag
  * That produces an object file, `run_utils.o`
* Error handling:
  * Error handling is always the *FIRST* thing you do in a function
  * Use an enumerated type to remove "magic numbers"
  * You generally check for null pointer *first*
  * Zero indicates *no error*, non-zero indicates some *type* of error
* Testing:
  * You should write formal/informal unit tests to give some evidence that your
    code is correct
  * You should come up with a input-output pair that is known to be correct using
  *some other means* (NOT your program)
  * You can write an informal testing suite OR use/adapt a formal unit testing framework (cmocka)
  * Compiling: `gcc run_utils_tester.c run_utils.o -lm -Wall -Wextra`

```c
typedef enum {
  NO_ERROR,
  NEGATIVE_VELOCITY_ERROR,
  NULL_POINTER_ERROR,
} ErrorCode;

#define KMS_PER_HOUR_TO_MILES_PER_HOUR 0.62137119
#define KMS_PER_HOUR_TO_FT_PER_SECOND 0.91134442
#define KMS_PER_HOUR_TO_M_PER_SECOND 0.27777778


/**
 * Converts the given kmsPerHour velocity to the other three scales/units.
 *
 * Returns a non-zero error code on any error.
 */
int convert(double kmsPerHour, double *milesPerHour, double *feetPerSecond, double *metersPerSecond);

```

# `run_utils.c`

```c
#include <stdlib.h>

#include "run_utils.h"

int convert(double kmsPerHour, double *milesPerHour, double *feetPerSecond, double *metersPerSecond) {

    if(milesPerHour == NULL || feetPerSecond == NULL || metersPerSecond == NULL) {
        return NULL_POINTER_ERROR;
    } else if(kmsPerHour < 0) {
        return NEGATIVE_VELOCITY_ERROR;
    }

    *milesPerHour    = kmsPerHour * KMS_PER_HOUR_TO_MILES_PER_HOUR;
    *feetPerSecond   = kmsPerHour * KMS_PER_HOUR_TO_FT_PER_SECOND;
    *metersPerSecond = kmsPerHour * KMS_PER_HOUR_TO_M_PER_SECOND;

    return NO_ERROR;
}

```

```c
/**

#include <stdlib.h>
#include <stdio.h>
#include <math.h>

#include "run_utils.h"

int main(int argc, char **argv) {

    double const EPSILON = 0.0001;

    int numFailed = 0;
    int numPassed = 0;

    double kmsPerHourInput = 9.8;
    double milesPerHourActual;
    double feetPerSecondActual;
    double metersPerSecondActual;

    double milesPerHourExpected;
    double feetPerSecondExpected;
    double metersPerSecondExpected;

    milesPerHourExpected = 6.0894377;
    feetPerSecondExpected = 8.9311753;
    metersPerSecondExpected = 2.7222222;

    convert(kmsPerHourInput, &milesPerHourActual, &feetPerSecondActual, &metersPerSecondActual );

    //TODO: modify to also check the error code, it should be zero...
    if( fabs(milesPerHourExpected - milesPerHourActual) > EPSILON) {
        printf("FAILED: expected %.20f but got %.20f\n", milesPerHourExpected, milesPerHourActual);
        numFailed++;
    } else if( fabs(feetPerSecondExpected - feetPerSecondActual) > EPSILON) {
        printf("FAILED: expected %.20f but got %.20f\n", feetPerSecondExpected, feetPerSecondActual);
        numFailed++;
    } else if( fabs(metersPerSecondExpected - metersPerSecondActual) > EPSILON) {
        printf("FAILED: expected %.20f but got %.20f\n", metersPerSecondExpected, metersPerSecondActual);
        numFailed++;
    } else {
        printf("PASSED\n");
        numPassed++;
    }

    kmsPerHourInput = -9.8;
    int errorCodeActual = convert(kmsPerHourInput, &milesPerHourActual, &feetPerSecondActual, &metersPerSecondActual);
    int errorCodeExpected = NEGATIVE_VELOCITY_ERROR;
    if(errorCodeActual != errorCodeExpected) {
        printf("FAILED: error code - expected %d, but received %d\n", errorCodeExpected, errorCodeActual);
        numFailed++;
    } else {
        printf("PASSED\n");
        numPassed++;
    }

    kmsPerHourInput = 5.8;
    errorCodeActual = convert(kmsPerHourInput, NULL, &feetPerSecondActual, &metersPerSecondActual);
    errorCodeExpected = NULL_POINTER_ERROR;
    if(errorCodeActual != errorCodeExpected) {
        printf("FAILED: error code - expected %d, but received %d\n", errorCodeExpected, errorCodeActual);
        numFailed++;
    } else {
        printf("PASSED\n");
        numPassed++;
    }




    return 0;
}
```
